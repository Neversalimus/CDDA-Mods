cmake_minimum_required(VERSION 3.20)
project(AdvancedWorldSettings LANGUAGES CXX)
if(NOT EXISTS "${NCMM_SDK_DIR}/ncmm_api.h")
  message(FATAL_ERROR "Set NCMM_SDK_DIR to the pinned external NCMM sdk directory")
endif()
add_library(ncmm_mod SHARED src/aws.cpp)
target_compile_features(ncmm_mod PRIVATE cxx_std_17)
target_include_directories(ncmm_mod PRIVATE "${NCMM_SDK_DIR}")
if(MSVC)
  set_property(TARGET ncmm_mod PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ncmm_mod PRIVATE /EHsc)
endif()
set_target_properties(ncmm_mod PROPERTIES PREFIX "" OUTPUT_NAME "ncmm_mod")
